class Solution {
    public boolean isPalindrome(int x) {
     if(x<0)
     return false;
      int o=x;
        int b,r=0;
        while(x>0){
            b=x%10;
            r=(r*10)+b;
            x=x/10;
        }
        if(r==o)
        return true;
        else
        return false;
     }    
}
